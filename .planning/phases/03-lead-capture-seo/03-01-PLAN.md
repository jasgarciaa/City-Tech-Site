STEP 3 — Component signature and state:
```typescript
interface ServiceRequestFormProps {
  services: ServiceDefinition[]
}

export default function ServiceRequestForm({ services }: ServiceRequestFormProps) {
  const [submitState, setSubmitState] = useState<SubmitState>('idle')

  const form = useForm<FormValues>({
    resolver: zodResolver(schema),
    defaultValues: {
      name: '',
      phone: '',
      email: '',
      services_requested: [],
      fleet: false,
      vehicle_make: '',
      vehicle_model: '',
      vehicle_year: '',
      description: '',
      preferred_datetime: '',
      vin: '',
      referral_source: '',
    },
  })

  const electronics = services
    .filter(s => s.category === 'electronics')
    .sort((a, b) => a.order - b.order)
  const locksmith = services
    .filter(s => s.category === 'locksmith')
    .sort((a, b) => a.order - b.order)

  const selectedServices = form.watch('services_requested')

  function toggleService(name: string, checked: boolean) {
    const current = form.getValues('services_requested')
    if (checked) {
      form.setValue('services_requested', [...current, name], { shouldValidate: true })
    } else {
      form.setValue(
        'services_requested',
        current.filter(s => s !== name),
        { shouldValidate: true }
      )
    }
  }

  async function onSubmit(values: FormValues) {
    setSubmitState('loading')
    try {
      // D-25 / Pitfall 3: fleet is NOT sent to server. If true, prepend to description.
      const baseDescription = values.description?.trim() ?? ''
      const description = values.fleet
        ? baseDescription
          ? `Fleet customer: Yes\n${baseDescription}`
          : 'Fleet customer: Yes'
        : baseDescription || undefined

      await submitServiceRequest({
        name: values.name.trim(),
        phone: values.phone.trim(),
        email: values.email.trim(),
        services_requested: values.services_requested,
        vehicle_make: values.vehicle_make.trim(),
        vehicle_model: values.vehicle_model.trim(),
        vehicle_year: values.vehicle_year.trim(),
        description,
        preferred_datetime: values.preferred_datetime?.trim() || undefined,
        vin: values.vin?.trim() || undefined,
        referral_source: values.referral_source?.trim() || undefined,
      })
      setSubmitState('success')
    } catch {
      setSubmitState('error')
    }
  }
```
